# MailMoose — curl cookbook
#
# Copy-pasteable REST scenarios using only curl and jq. Every request carries
# `Authorization: Bearer <key>`.
#
# PREREQUISITE: this cookbook uses curl and jq. If jq is not on PATH, drop the
# `| jq .` pipes (the raw response is already JSON) or use the stdlib-only
# Python client at `GET /examples/python`, which needs neither.
#
# export BASE="https://your-instance" # origin serving this cookbook
# export KEY="mmm_..."
#   export INBOX="inb_..."      # set after the bootstrap step below
#   export HELP='-H "Authorization: Bearer $KEY"'   # optional shell shorthand
#
# Responses are JSON; piping through `jq .` pretty-prints them where jq is
# available. Add `-i` to any request to see the status line and headers.
#
# Prefer the served clients: `GET /examples/python` (mailmoose.py, standard
# library only) and `GET /examples/bash` (mailmoose.sh, requires curl and jq).
# Each is one command per operation, sends the JSON body inline, and prints the
# response to stdout. This cookbook exists so the same calls can be made by hand
# or from another language.
#
# Safe-calling rules (inline single-command bodies, no pipe, no file staging)
# are in `GET /agent`. Keep each call below a single command with an inline body
# and no formatter pipe.

# ---------------------------------------------------------------------------
# 0. Discovery (no auth required)
# ---------------------------------------------------------------------------

curl -sS "$BASE/.well-known/mailmoose" | jq .
curl -sS "$BASE/agent"                      # Markdown usage guide
curl -sS "$BASE/openapi.json" | jq .        # OpenAPI document
curl -sS "$BASE/examples/bash"              # served Bash client (mailmoose.sh)
curl -sS "$BASE/examples/python"            # served Python client (mailmoose.py)

# ---------------------------------------------------------------------------
# 1. Bootstrap — who am I, which inboxes can I reach
# ---------------------------------------------------------------------------

curl -sS -H "Authorization: Bearer $KEY" "$BASE/v1/bootstrap" | jq .

INBOX=$(curl -sS -H "Authorization: Bearer $KEY" "$BASE/v1/bootstrap" \
  | jq -r '.inboxes[0].id')

# Equivalent direct inbox listing:
curl -sS -H "Authorization: Bearer $KEY" "$BASE/v1/inboxes" | jq .

# ---------------------------------------------------------------------------
# 2. List messages with filters
# ---------------------------------------------------------------------------

# Newest unread messages in an inbox, at most 20.
curl -sS -H "Authorization: Bearer $KEY" \
  "$BASE/v1/messages?inbox=$INBOX&unread=true&limit=20" | jq .

# Messages carrying BOTH labels (repeated label params are ANDed), with an
# attachment, sent by a given address, older than a keyset cursor.
curl -sS -H "Authorization: Bearer $KEY" \
  "$BASE/v1/messages?inbox=$INBOX&label=Invoices&label=Unpaid&has_attachment=true&from=jane@example.org&before=msg_01K..." \
  | jq .

# ---------------------------------------------------------------------------
# 3. Get one message
# ---------------------------------------------------------------------------

MSG="msg_01K..."
curl -sS -H "Authorization: Bearer $KEY" "$BASE/v1/messages/$MSG" | jq .

# Attachment metadata, then the bytes of one attachment.
curl -sS -H "Authorization: Bearer $KEY" \
  "$BASE/v1/messages/$MSG/attachments" | jq .
ATT="att_01K..."
curl -sS -H "Authorization: Bearer $KEY" \
  -o quote.pdf "$BASE/v1/attachments/$ATT"

# ---------------------------------------------------------------------------
# 4. Full-text search (SQLite FTS5)
# ---------------------------------------------------------------------------

curl -sS -H "Authorization: Bearer $KEY" \
  "$BASE/v1/search?q=invoice&inbox=$INBOX&limit=20" | jq .

# Search also accepts the usual filters: label, from, to, has_attachment.
curl -sS -H "Authorization: Bearer $KEY" \
  "$BASE/v1/search?q=quote&inbox=$INBOX&has_attachment=true" | jq .

# ---------------------------------------------------------------------------
# 5. Send
# ---------------------------------------------------------------------------

# The canonical shape: ONE curl command, body inline, and nothing after it.
# Do not stage the body in a file, pass `--data-binary @file`, or pipe the
# response to a formatter. Upload-shaped and chained-execution commands are
# flagged by host security scanners and then require human approval.
curl -sS -X POST -H "Authorization: Bearer $KEY" -H "Content-Type: application/json" \
  -d "{\"inbox_id\":\"$INBOX\",\"to\":[\"alice@example.org\"],\"subject\":\"Report\",\"text\":\"See the report.\"}" \
  "$BASE/v1/send"

# Idempotency-Key makes the send retry-safe: re-sending the same key returns the
# already-created message instead of sending twice. Block until delivery instead
# of returning as soon as it is queued with `?wait=true`.
IDEM=$(uuidgen 2>/dev/null || printf 'gh-%s-%s' "$(date +%s)" "$$")
curl -sS -X POST -H "Authorization: Bearer $KEY" -H "Content-Type: application/json" \
  -H "Idempotency-Key: $IDEM" \
  -d "{\"inbox_id\":\"$INBOX\",\"to\":[\"alice@example.org\"],\"subject\":\"Report\",\"text\":\"See the report.\"}" \
  "$BASE/v1/send?wait=true"

# With an attachment, the base64 payload makes a jq-built body the practical
# option. This shape needs jq; without it, use `GET /examples/python`, whose
# `send --attach` does the same encoding.
BODY=$(jq -n \
  --arg inbox "$INBOX" \
  --arg b64 "$(base64 -w0 report.pdf)" \
  '{
     inbox_id: $inbox,
     to: ["alice@example.org"],
     subject: "Report",
     text: "See the attached report.",
     attachments: [
       {filename: "report.pdf", content_type: "application/pdf", content: $b64}
     ]
   }')
curl -sS -X POST -H "Authorization: Bearer $KEY" -H "Content-Type: application/json" \
  -H "Idempotency-Key: $IDEM-attach" \
  -d "$BODY" "$BASE/v1/send"

# ---------------------------------------------------------------------------
# 6. Reply to a message (In-Reply-To/References are set server-side)
# ---------------------------------------------------------------------------

curl -sS -X POST -H "Authorization: Bearer $KEY" -H "Content-Type: application/json" \
  -H "Idempotency-Key: reply-$IDEM" \
  -d '{"text":"Thanks for the note!"}' \
  "$BASE/v1/messages/$MSG/reply" | jq .

# ---------------------------------------------------------------------------
# 7. Patch message state and labels
# ---------------------------------------------------------------------------

# Mark read / unread.
curl -sS -X PATCH -H "Authorization: Bearer $KEY" -H "Content-Type: application/json" \
  -d '{"read":true}' "$BASE/v1/messages/$MSG" | jq .

# Replace the label set (`[]` clears every label).
curl -sS -X PATCH -H "Authorization: Bearer $KEY" -H "Content-Type: application/json" \
  -d '{"labels":["Invoices","Unpaid"]}' "$BASE/v1/messages/$MSG" | jq .

# Quarantine / release from Spam.
curl -sS -X PATCH -H "Authorization: Bearer $KEY" -H "Content-Type: application/json" \
  -d '{"spam":true}' "$BASE/v1/messages/$MSG" | jq .

# Delete a message (Assistant/Owner).
curl -sS -X DELETE -H "Authorization: Bearer $KEY" "$BASE/v1/messages/$MSG"

# ---------------------------------------------------------------------------
# 8. Drafts
# ---------------------------------------------------------------------------

# List drafts in an inbox.
curl -sS -H "Authorization: Bearer $KEY" "$BASE/v1/drafts?inbox=$INBOX" | jq .

# Create a draft (action defaults to "draft").
DRAFT_BODY=$(jq -n --arg inbox "$INBOX" \
  '{inbox_id:$inbox, to:["alice@example.org"], subject:"Quote", text:"Draft body"}')
curl -sS -X POST -H "Authorization: Bearer $KEY" -H "Content-Type: application/json" \
  -d "$DRAFT_BODY" "$BASE/v1/drafts" | jq .

DRAFT=$(curl -sS -X POST -H "Authorization: Bearer $KEY" -H "Content-Type: application/json" \
  -d "$DRAFT_BODY" "$BASE/v1/drafts" | jq -r '.id')

# Get / partial-update / delete a draft.
curl -sS -H "Authorization: Bearer $KEY" "$BASE/v1/drafts/$DRAFT" | jq .
curl -sS -X PATCH -H "Authorization: Bearer $KEY" -H "Content-Type: application/json" \
  -d '{"subject":"Quote (revised)"}' "$BASE/v1/drafts/$DRAFT" | jq .
curl -sS -X DELETE -H "Authorization: Bearer $KEY" "$BASE/v1/drafts/$DRAFT"

# Create + inline-attach + submit for approval in a single request:
#   {"action":"request-send"} (Assistant) or {"action":"send"} (Owner).

# ---------------------------------------------------------------------------
# 9. Multipart draft attachment upload (field name is exactly "attachments")
# ---------------------------------------------------------------------------

curl -sS -X POST -H "Authorization: Bearer $KEY" \
  -F "attachments=@report.pdf;type=application/pdf" \
  "$BASE/v1/drafts/$DRAFT/attachments" | jq .

# List, download, delete one draft attachment (second upload shown for the id).
ATT2=$(curl -sS -X POST -H "Authorization: Bearer $KEY" \
  -F "attachments=@report.pdf;type=application/pdf" \
  "$BASE/v1/drafts/$DRAFT/attachments" | jq -r '.[0].id')
curl -sS -H "Authorization: Bearer $KEY" "$BASE/v1/drafts/$DRAFT/attachments" | jq .
curl -sS -H "Authorization: Bearer $KEY" \
  -o downloaded.pdf "$BASE/v1/drafts/$DRAFT/attachments/$ATT2"
curl -sS -X DELETE -H "Authorization: Bearer $KEY" \
  "$BASE/v1/drafts/$DRAFT/attachments/$ATT2"

# ---------------------------------------------------------------------------
# 10. Draft approval workflow (Assistant requests, Owner authorizes)
# ---------------------------------------------------------------------------

# Assistant: submit the frozen draft for authorization.
curl -sS -X POST -H "Authorization: Bearer $KEY" -H "Content-Type: application/json" \
  -d '{"external":true}' \
  "$BASE/v1/drafts/$DRAFT/request-send" | jq .

# Inspect requests (active only, for one inbox).
curl -sS -H "Authorization: Bearer $KEY" \
  "$BASE/v1/send-requests?inbox=$INBOX&active=true" | jq .

# The latest request for a draft (works even after the draft was sent).
curl -sS -H "Authorization: Bearer $KEY" "$BASE/v1/drafts/$DRAFT/send-request" | jq .

# Owner: approve and enqueue (approval and delivery are separate states).
curl -sS -X POST -H "Authorization: Bearer $KEY" -H "Content-Type: application/json" \
  -H "Idempotency-Key: approve-$IDEM" \
  -d '{"feedback":"Looks good"}' \
  "$BASE/v1/drafts/$DRAFT/approve" | jq .

# Owner: reject with feedback (draft stays editable).
curl -sS -X POST -H "Authorization: Bearer $KEY" -H "Content-Type: application/json" \
  -d '{"feedback":"Please revise the numbers"}' \
  "$BASE/v1/drafts/$DRAFT/reject" | jq .

# Assistant: withdraw a pending request.
curl -sS -X POST -H "Authorization: Bearer $KEY" \
  "$BASE/v1/drafts/$DRAFT/cancel-send-request" | jq .

# Direct send of a draft (Owner) instead of the approval flow.
curl -sS -X POST -H "Authorization: Bearer $KEY" -H "Content-Type: application/json" \
  -d '{}' "$BASE/v1/drafts/$DRAFT/send" | jq .

# ---------------------------------------------------------------------------
# 11. Outbox — pending/failed sends, retry, cancel
# ---------------------------------------------------------------------------

curl -sS -H "Authorization: Bearer $KEY" "$BASE/v1/outbox?inbox=$INBOX" | jq .
OUT="omsg_01K..."
curl -sS -X POST -H "Authorization: Bearer $KEY" "$BASE/v1/outbox/$OUT/retry"
curl -sS -X DELETE -H "Authorization: Bearer $KEY" "$BASE/v1/outbox/$OUT"

# ---------------------------------------------------------------------------
# 12. Events — one page of durable history
# ---------------------------------------------------------------------------

curl -sS -H "Authorization: Bearer $KEY" \
  "$BASE/v1/events?after=evt_123&inbox=$INBOX&limit=100" | jq .

# ===========================================================================
# 13. SSE stream — NON-RETURNING
# ---------------------------------------------------------------------------
# Opens the event stream and keeps the connection open until interrupted
# (Ctrl-C). `curl -N` disables output buffering so events appear immediately.
# Backlog after `after` is delivered first, then live events. This does NOT
# return on its own; run it on a dedicated worker, not inside a request.
# ===========================================================================

curl -N -H "Authorization: Bearer $KEY" -H "Accept: text/event-stream" \
  "$BASE/v1/events/stream?after=evt_123&inbox=$INBOX"

# Each event is emitted as:
#   id: evt_123
#   event: message.received
#   data: {"cursor":"evt_123","type":"message.received", ...}

# ===========================================================================
# 14. Long poll — BLOCKS up to --timeout seconds
# ---------------------------------------------------------------------------
# Calls the JSON long-poll endpoint and blocks until a new message arrives or
# the timeout (server cap 600s) elapses. On timeout the server returns 204 with
# no body. Prefer `events` for a single non-blocking poll.
# ===========================================================================

curl -sS -H "Authorization: Bearer $KEY" \
  "$BASE/v1/messages/wait?inbox=$INBOX&after=evt_123&timeout=60" | jq .

# 204 No Content means no new message arrived before the timeout.
curl -sS -o /dev/null -w '%{http_code}\n' -H "Authorization: Bearer $KEY" \
  "$BASE/v1/messages/wait?inbox=$INBOX&timeout=60"

# ===========================================================================
# 15. Common mailbox surface — folders and messages (domain or standalone)
# ---------------------------------------------------------------------------
# These endpoints dispatch on the inbox kind, so the same calls read a managed
# domain inbox (local store) and a standalone inbox (live IMAP). Every listing
# returns the shared envelope {items, next_cursor, completeness, errors}.
# ===========================================================================

# List an inbox's folder tree (hierarchical). `errors` carries a per-inbox
# failure and `completeness` is "unknown" when a remote server is unreachable but
# the last-known folders are still served.
curl -sS -H "Authorization: Bearer $KEY" \
  "$BASE/v1/inboxes/$INBOX/folders" | jq '{folders: [.items[].path], completeness, errors}'

# Create a folder (Assistant or Owner). A standalone inbox creates it on the
# live server; a domain inbox creates it locally.
curl -sS -X POST -H "Authorization: Bearer $KEY" -H "Content-Type: application/json" \
  -d '{"path":"Archive/2026"}' "$BASE/v1/inboxes/$INBOX/folders" | jq .

# List messages in one folder (a folder path for a standalone inbox, a folder id
# for a domain inbox). `before` is the opaque cursor from the previous page.
curl -sS -H "Authorization: Bearer $KEY" \
  "$BASE/v1/inboxes/$INBOX/messages?folder=INBOX&limit=50" | jq '{items: [.items[].subject], next_cursor, completeness}'

# Download a message's raw MIME as an attachment (local file, or streamed live
# from the remote server for a standalone inbox; never archived).
curl -sS -H "Authorization: Bearer $KEY" \
  "$BASE/v1/inboxes/$INBOX/messages/$MSG/content" -o message.eml

# Download one MIME attachment (a numeric part path, e.g. 1.2 for a nested part).
curl -sS -H "Authorization: Bearer $KEY" \
  "$BASE/v1/inboxes/$INBOX/messages/$MSG/attachments/1.2?filename=report.pdf" -o report.pdf

# ===========================================================================
# 16. Standalone inbox remote connector (IMAP/SMTP)
# ---------------------------------------------------------------------------
# The non-secret remote description is readable by anyone who can read the
# inbox; the password fields are only ever reported present/absent, never
# returned. Configure/test/refresh require Owner.
# ===========================================================================

curl -sS -H "Authorization: Bearer $KEY" \
  "$BASE/v1/inboxes/$INBOX/remote" | jq '{host, port, security, configured, imap_password_set, missing_roles}'

# Save the connector. A blank secret keeps the stored one. security defaults to
# tls; starttls and plain are explicit choices (plain may be refused by policy).
curl -sS -X PUT -H "Authorization: Bearer $KEY" -H "Content-Type: application/json" \
  -d '{"host":"imap.example.com","port":993,"username":"agent@example.com","security":"tls","namespace":"INBOX","imap_password":"app-password"}' \
  "$BASE/v1/inboxes/$INBOX/remote" | jq '{status: "saved"}'

# Test the connector without saving (a blank imap_password uses the stored one).
curl -sS -X POST -H "Authorization: Bearer $KEY" -H "Content-Type: application/json" \
  -d '{}' "$BASE/v1/inboxes/$INBOX/remote/test" | jq .

# Force a full metadata reconcile (folders and headers; bodies are never stored).
curl -sS -X POST -H "Authorization: Bearer $KEY" \
  "$BASE/v1/inboxes/$INBOX/remote/refresh" | jq .

# ===========================================================================
# 17. Merged pagination, filters, role mapping, sent copy, remote purge
# ===========================================================================
# An account-wide listing merges the local store and every authorized standalone
# inbox into ONE globally date-sorted stream. The cursor is opaque: pass the
# next_cursor value straight back as before.
curl -sS -H "Authorization: Bearer $KEY" \
  "$BASE/v1/messages?limit=50" | jq '{subjects: [.items[].subject], next_cursor, errors}'
# Follow the cursor (repeat until next_cursor is empty).
curl -sS -H "Authorization: Bearer $KEY" \
  "$BASE/v1/messages?limit=50&before=$CURSOR" | jq .

# Search filters are applied on both local and remote mail: from/to/subject/
# unread/has_attachment/label, plus folder. An unsupported combination is reported.
curl -sS -H "Authorization: Bearer $KEY" \
  "$BASE/v1/search?q=invoice&unread=true&has_attachment=true" | jq '{items: [.items[].subject], completeness}'

# Map an EXISTING (arbitrarily named) remote folder to a special role by id.
curl -sS -X POST -H "Authorization: Bearer $KEY" -H "Content-Type: application/json" \
  -d '{"folder_id":"fld_123"}' "$BASE/v1/inboxes/$INBOX/remote/roles/trash" | jq .
# Or create the conventional folder for the role.
curl -sS -X POST -H "Authorization: Bearer $KEY" -H "Content-Type: application/json" \
  -d '{"create":true}' "$BASE/v1/inboxes/$INBOX/remote/roles/trash" | jq .

# Sent-copy switch (standalone): copy a sent message into the remote Sent folder.
curl -sS -X PUT -H "Authorization: Bearer $KEY" -H "Content-Type: application/json" \
  -d '{"sent_copy_enabled":true,"sent_copy_folder":""}' \
  "$BASE/v1/inboxes/$INBOX/remote" | jq '{sent_copy_enabled, sent_copy_folder}'

# Permanent remote purge (Owner): the message must already be in the mapped
# Trash-role folder. Trash first, then purge.
curl -sS -X DELETE -H "Authorization: Bearer $KEY" "$BASE/v1/messages/$MSG"
curl -sS -X DELETE -H "Authorization: Bearer $KEY" "$BASE/v1/messages/$MSG/purge"

# ===========================================================================
# 18. Assistant authoring: mode and RemoteDraft handoff history
# ===========================================================================
# The effective authoring mode decides how a send request is handled. approver_
# enabled tracks the EFFECTIVE mode, not the inbox kind: a standalone inbox
# switched to mailmoose_approval enables the approver. remote_draft is a
# standalone-only mode — a domain inbox is preset to mailmoose_approval and
# rejects remote_draft with 400. standalone is true for a remote (IMAP/SMTP)
# mailbox.
curl -sS -H "Authorization: Bearer $KEY" \
  "$BASE/v1/inboxes/$INBOX/authoring" | jq '{mode, default_mode, approver_enabled, notify_address, standalone}'

# Switch a standalone inbox's mode (Owner). Empty mode clears to the kind default.
curl -sS -X PUT -H "Authorization: Bearer $KEY" -H "Content-Type: application/json" \
  -d '{"mode":"remote_draft"}' "$BASE/v1/inboxes/$INBOX/authoring" | jq .

# A RemoteDraft handoff is one-way to the connected remote Drafts folder. The
# history retains terminal records (published, ambiguous, failed/cancelled) even
# after the local draft is cleaned up, and reports publication and notification
# independently.
curl -sS -H "Authorization: Bearer $KEY" \
  "$BASE/v1/inboxes/$INBOX/handoffs" | jq '{handoffs: [.items[] | {draft_id, publication, notification_status, remote_folder, last_error}]}'
